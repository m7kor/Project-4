cmake_minimum_required(VERSION 3.24)
project(re4_runtime_libraries LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(RE4_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DEPS "${RE4_ROOT}/dependencies/runtime-libraries")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(DXHEADERS_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(DXHEADERS_BUILD_GOOGLE_TEST OFF CACHE BOOL "" FORCE)
set(DXHEADERS_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${RE4_ROOT}/core/deps/raylib" raylib)
add_subdirectory("${DEPS}/DirectX-Headers" directx)
add_library(imgui STATIC
  "${DEPS}/imgui/imgui.cpp" "${DEPS}/imgui/imgui_draw.cpp"
  "${DEPS}/imgui/imgui_tables.cpp" "${DEPS}/imgui/imgui_widgets.cpp")
target_include_directories(imgui PUBLIC "${DEPS}/imgui")
add_library(rlImGui STATIC "${DEPS}/rlImGui/rlImGui.cpp")
target_include_directories(rlImGui PUBLIC "${DEPS}/rlImGui")
target_link_libraries(rlImGui PUBLIC imgui raylib)
add_library(ffmpeg INTERFACE)
target_include_directories(ffmpeg INTERFACE "${DEPS}/ffmpeg/include")
foreach(component avcodec avformat avutil swresample swscale)
  if(NOT EXISTS "${DEPS}/ffmpeg/bin/${component}.lib")
    message(FATAL_ERROR "Missing local FFmpeg import library: ${component}")
  endif()
  target_link_libraries(ffmpeg INTERFACE "${DEPS}/ffmpeg/bin/${component}.lib")
endforeach()
add_executable(runtime_libraries_check main.cpp raylib_check.cpp imgui_check.cpp ffmpeg_check.cpp directx_check.cpp)
target_link_libraries(runtime_libraries_check PRIVATE rlImGui ffmpeg Microsoft::DirectX-Headers Microsoft::DirectX-Guids)
file(GLOB FFMPEG_DLLS "${DEPS}/ffmpeg/bin/*.dll")
add_custom_command(TARGET runtime_libraries_check POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "$<TARGET_FILE_DIR:runtime_libraries_check>"
  COMMAND_EXPAND_LISTS)
