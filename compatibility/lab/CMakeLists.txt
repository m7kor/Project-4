cmake_minimum_required(VERSION 3.21)
project(RE4Compatibility LANGUAGES CXX)
enable_testing()
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RUNTIME "${ROOT}/engines/project4/ps2xRuntime")
add_executable(gif_image2_before "${ROOT}/tests/gif_image2_test.cpp" "${ROOT}/artifacts/gif-before/gs/ps2_gif_arbiter.cpp")
add_executable(gif_image2_after "${ROOT}/tests/gif_image2_test.cpp" "${RUNTIME}/src/lib/gs/ps2_gif_arbiter.cpp")
foreach(t gif_image2_before gif_image2_after)
 target_include_directories(${t} PRIVATE "${RUNTIME}/include")
 target_compile_features(${t} PRIVATE cxx_std_20)
endforeach()
add_test(NAME image2_pre_fix_must_fail COMMAND gif_image2_before)
set_tests_properties(image2_pre_fix_must_fail PROPERTIES WILL_FAIL TRUE)
add_test(NAME image2_corrected COMMAND gif_image2_after)
add_subdirectory("${ROOT}/engines/project4/ps2xTest/gs_cache" gs-cache)
add_executable(image2_frontend "${ROOT}/tests/image2_frontend_test.cpp")
target_link_libraries(image2_frontend PRIVATE ps2_gs_cache_backend_under_test)
add_test(NAME image2_frontend_vram COMMAND image2_frontend)
